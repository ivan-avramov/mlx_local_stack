+// maxInt64 is the largest value an int64 can hold.
+const maxInt64 = int64(1<<63 - 1)
+
+// ParseHex converts a hexadecimal string (e.g. "10af8c") to its decimal
+// equivalent using first principles, without any library conversion.
+// It returns an error whose message contains "syntax" for invalid input
+// and "range" when the value does not fit in an int64.
+func ParseHex(in string) (int64, error) {
+	if in == "" {
+		return 0, errSyntax
+	}
+
+	var n int64
+	for _, c := range in {
+		var d int64
+		switch {
+		case c >= '0' && c <= '9':
+			d = int64(c - '0')
+		case c >= 'a' && c <= 'f':
+			d = int64(c-'a') + 10
+		case c >= 'A' && c <= 'F':
+			d = int64(c-'A') + 10
+		default:
+			return 0, errSyntax
+		}
+		// Check for overflow before folding this digit in.
+		if n > (maxInt64-d)/16 {
+			return 0, errRange
+		}
+		n = n*16 + d
+	}
+	return n, nil
 }
+
+// HandleErrors takes a list of inputs for ParseHex and returns a matching
+// list of error cases: "none", "syntax", or "range".
+func HandleErrors(inputs []string) []string {
+	results := make([]string, len(inputs))
+	for i, in := range inputs {
+		_, err := ParseHex(in)
+		switch {
+		case err == nil:
+			results[i] = "none"
+		case errors.Is(err, errRange):
+			results[i] = "range"
+		default:
+			results[i] = "syntax"
+		}
+	}
