+func ParseHex(in string) (int64, error) {
+	if len(in) == 0 {
+		return 0, errSyntax

+
+	var out int64
+	for _, c := range in {
+		var digit int64
+		switch {

+			digit = int64(c - '0')

+			digit = int64(c-'a') + 10

+		if out > (0x7FFFFFFFFFFFFFFF-digit)/16 {
