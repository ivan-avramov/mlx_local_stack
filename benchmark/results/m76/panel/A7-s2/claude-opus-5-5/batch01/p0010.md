+func ParseHex(in string) (int64, error) {
+	if in == "" {
+		return 0, fmt.Errorf("syntax error: empty string")
+	}
+
+	var out int64
+	for _, c := range in {
+		var digit int64
+		switch {
+		case c >= '0' && c <= '9':
+			digit = int64(c - '0')
+		case c >= 'a' && c <= 'f':
+			digit = int64(c-'a') + 10
+		case c >= 'A' && c <= 'F':
+			digit = int64(c-'A') + 10
+		default:
+			return 0, fmt.Errorf("syntax error: %q is not a hexadecimal digit", c)
+		}
+
+		// out*16+digit must not overflow int64.
+		if out > (math.MaxInt64-digit)/16 {
+			return 0, fmt.Errorf("range error: value out of range")
+		}
+		out = out*16 + digit
+	}

 }
